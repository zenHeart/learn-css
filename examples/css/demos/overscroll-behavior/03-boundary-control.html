代码示例
/* 阻止滚动链，保留弹性效果 */
.dialog-content {
  overflow-y: auto;
  overscroll-behavior: contain;
}

/* 彻底禁用边界效果 */
.scroll-area {
  overscroll-behavior: none;
}

/* 分轴控制 */
.carousel {
  overscroll-behavior-x: contain;
  overscroll-behavior-y: none;
}
