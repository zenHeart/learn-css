React Native 对应实现
CSS → React Native 映射
// CSS
display: flex;
flex-wrap: wrap;
gap: 10px;

// RN (React Native)
display: 'flex';
flexWrap: 'wrap';
gap: 10;

// 注意：gap 在 RN 0.76+ 支持
完整 RN 包裹布局示例
import { View, StyleSheet } from 'react-native';

const WrapDemo = () => (
  View style={styles.container}>
    View style={styles.item}>A</View>
    View style={styles.item}>B</View>
    View style={styles.item}>C</View>
  </View>
);

const styles = StyleSheet.create({
  container: {
    display: 'flex',
    flexDirection: 'row',
    flexWrap: 'wrap',
    gap: 10,
  },
  item: {
    width: 100,
    height: 60,
    backgroundColor: '#667eea',
    borderRadius: 6,
  },
});
