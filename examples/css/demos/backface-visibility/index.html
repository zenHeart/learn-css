Basic Implementation

.card-container {
  perspective: 1000px;
}

.card {
  position: relative;
  transform-style: preserve-3d;
  transition: transform 0.6s;
}

.card-face {
  position: absolute;
  backface-visibility: hidden;
}

.card-front {
  /* Front face styles */
}

.card-back {
  transform: rotateY(180deg);
  /* Back face styles */
}
