/* 父网格定义命名线 */ .parent { display: grid; grid-template-columns: [full-start] 1fr [content-start] 200px [content-end] 1fr [full-end]; } /* 子网格继承命名线 */ .child { display: grid; grid-template-columns: subgrid; /* 可以直接使用 full-start, content-start 等 */ } .child-header { grid-column: full-start / full-end; } .child-body { grid-column: content-start / content-end; }
