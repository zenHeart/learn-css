/* 父网格 */ .parent { display: grid; gap: 16px; } /* 子网格继承 gap */ .child { display: grid; grid-template-columns: subgrid; /* gap: 16px 继承 */ } /* 子网格覆盖 gap */ .child { display: grid; grid-template-columns: subgrid; gap: 8px; /* 覆盖 */ } /* 只覆盖列 gap */ .child { display: grid; grid-template-columns: subgrid; column-gap: 4px; /* 行 gap 保持 16px */ }
