核心代码

/* 定义容器 */ .card-wrapper { container-type: inline-size; container-name: card; } /* 默认样式 */ .card { display: flex; flex-direction: column; } .card-title { font-size: 1em; } /* 中等容器 */ @container card (width > 400px) { .card { flex-direction: row; } .card-image { width: 40%; } .card-title { font-size: 1.25em; } } /* 大容器 */ @container card (width > 700px) { .card-title { font-size: 1.8em; } }
