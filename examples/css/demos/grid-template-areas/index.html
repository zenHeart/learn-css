完整代码示例

/* 1. 定义网格容器 */ .page { display: grid; grid-template-columns: 200px 1fr; grid-template-rows: auto 1fr auto; grid-template-areas: "header header" "sidebar content" "footer footer"; min-height: 100vh; gap: 16px; } /* 2. 将元素分配到区域 */ .header { grid-area: header; background: #388bfd; } .sidebar { grid-area: sidebar; background: #a371f7; } .main { grid-area: content; background: #3fb950; } .footer { grid-area: footer; background: #f0883e; } /* 3. 响应式调整 */ @media (max-width: 768px) { .page { grid-template-columns: 1fr; grid-template-areas: "header" "main" "sidebar" "footer"; } }
